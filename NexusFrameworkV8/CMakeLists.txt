cmake_minimum_required(VERSION 3.20)
project(NexusV8Domain VERSION 0.2.0 LANGUAGES CXX)
option(NEXUS_SANITIZE "Address/undefined behavior sanitizer checks" OFF)
set(CMAKE_CXX_STANDARD 17 CACHE STRING "Portable C++ language level")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB DOMAIN_SOURCES CONFIGURE_DEPENDS Source/NexusV8Runtime/Private/Domain/*.cpp)
add_library(nexus_domain STATIC ${DOMAIN_SOURCES})
target_include_directories(nexus_domain PUBLIC Source/NexusV8Runtime/Public)
target_compile_definitions(nexus_domain PUBLIC NEXUS_STANDALONE=1)
if(MSVC)
  target_compile_options(nexus_domain PRIVATE /W4 /EHsc)
  target_compile_options(nexus_domain PUBLIC /utf-8)
else()
  target_compile_options(nexus_domain PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(NEXUS_SANITIZE AND NOT MSVC)
  target_compile_options(nexus_domain PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(nexus_domain PUBLIC -fsanitize=address,undefined)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/DomainTests.cpp")
  enable_testing()
  add_executable(nexus_tests Tests/DomainTests.cpp)
  target_link_libraries(nexus_tests PRIVATE nexus_domain)
  add_test(NAME nexus_domain COMMAND nexus_tests)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/PreviewExport.cpp")
  add_executable(nexus_preview Tests/PreviewExport.cpp)
  target_link_libraries(nexus_preview PRIVATE nexus_domain)
endif()
